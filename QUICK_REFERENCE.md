# 🚀 Quick Reference: Model Pattern + Fat Services

Quick copy-paste examples for common operations.

## 📁 File Structure

```
internal/
├── model/           # Domain models (your business types)
├── service/         # Fat services (business logic)
└── data/
    ├── db/          # Generated by sqlc
    ├── queries/     # Your SQL files
    └── migrations/  # Database schema
```

## 🎯 Common Patterns

### 1. Create a New Entity Type

```go
// internal/model/entity.go
package model

type Entity struct {
    ID        uuid.UUID `json:"id"`
    Name      string    `json:"name"`
    CreatedAt time.Time `json:"created_at"`
}

type CreateEntityInput struct {
    Name string `json:"name" validate:"required,min=1,max=255"`
}

type UpdateEntityInput struct {
    Name *string `json:"name,omitempty" validate:"omitempty,min=1"`
}
```

### 2. Create Service

```go
// internal/service/entity_service.go
package service

type EntityService struct {
    pool    *database.Pool
    queries *db.Queries
}

func NewEntityService(pool *database.Pool) *EntityService {
    return &EntityService{
        pool:    pool,
        queries: db.New(pool),
    }
}

func (s *EntityService) CreateEntity(ctx context.Context, input model.CreateEntityInput) (*model.Entity, error) {
    // Validate
    if input.Name == "" {
        return nil, fmt.Errorf("name required")
    }

    // Create in DB
    dbEntity, err := s.queries.CreateEntity(ctx, db.CreateEntityParams{
        ID:   uuid.New(),
        Name: input.Name,
    })
    if err != nil {
        return nil, err
    }

    // Convert to model
    return s.toEntityModel(dbEntity), nil
}

func (s *EntityService) toEntityModel(dbEntity db.Entity) *model.Entity {
    return &model.Entity{
        ID:        dbEntity.ID,
        Name:      dbEntity.Name,
        CreatedAt: dbEntity.CreatedAt,
    }
}
```

### 3. Create Handler

```go
// api/v1/entity/handler.go
package entity

import (
    "doit/internal/model"
    "doit/internal/service"
    "doit/internal/web"
)

type Handler struct {
    service *service.EntityService
}

func NewHandler(service *service.EntityService) *Handler {
    return &Handler{service: service}
}

func (h *Handler) CreateEntity(w http.ResponseWriter, r *http.Request) error {
    var input model.CreateEntityInput
    if err := web.DecodeJSON(r, &input); err != nil {
        return err
    }

    entity, err := h.service.CreateEntity(r.Context(), input)
    if err != nil {
        return err
    }

    return web.RespondCreated(w, r, entity)
}
```

## 📝 SQL Queries

### Create Query

```sql
-- internal/data/queries/entities.sql

-- name: CreateEntity :one
INSERT INTO entities (id, name)
VALUES ($1, $2)
RETURNING *;

-- name: GetEntityByID :one
SELECT * FROM entities WHERE id = $1;

-- name: ListEntities :many
SELECT * FROM entities
ORDER BY created_at DESC
LIMIT $1 OFFSET $2;

-- name: UpdateEntity :one
UPDATE entities
SET name = COALESCE($2, name)
WHERE id = $1
RETURNING *;

-- name: DeleteEntity :exec
DELETE FROM entities WHERE id = $1;
```

## 🔄 Common Operations

### Create with Metadata

```go
func (s *Service) Create(ctx, input model.Input) (*model.Entity, error) {
    metadata := input.Metadata
    if metadata == nil {
        metadata = make(map[string]interface{})
    }
    metadataJSON, _ := json.Marshal(metadata)

    entity, _ := s.queries.Create(ctx, db.CreateParams{
        ID:       uuid.New(),
        Metadata: metadataJSON,
    })

    return s.toModel(entity), nil
}
```

### Update with Optional Fields

```go
func (s *Service) Update(ctx, id uuid.UUID, input model.UpdateInput) (*model.Entity, error) {
    params := db.UpdateParams{ID: id}

    if input.Name != nil {
        params.Name = input.Name
    }

    entity, _ := s.queries.Update(ctx, params)
    return s.toModel(entity), nil
}
```

### Transaction

```go
func (s *Service) ComplexOperation(ctx, input) error {
    return database.WithTransaction(ctx, s.pool.Pool, database.DefaultTxOptions(), func(tx pgx.Tx) error {
        txQueries := s.queries.WithTx(tx)

        // Operation 1
        _, err := txQueries.Create(ctx, params1)
        if err != nil {
            return err // Auto-rollback
        }

        // Operation 2
        err = txQueries.Update(ctx, params2)
        return err // Auto-commit if nil
    })
}
```

### List with Pagination

```go
func (s *Service) List(ctx, limit, offset int32) ([]*model.Entity, error) {
    entities, err := s.queries.List(ctx, db.ListParams{
        Limit:  limit,
        Offset: offset,
    })
    if err != nil {
        return nil, err
    }

    return s.toModels(entities), nil
}
```

## 🎨 Type Conversions

### DB → Model (Single)

```go
func (s *Service) toModel(dbEntity db.Entity) *model.Entity {
    entity := &model.Entity{
        ID:        dbEntity.ID,
        Name:      dbEntity.Name,
        CreatedAt: dbEntity.CreatedAt,
    }

    // Nullable field
    if dbEntity.Description.Valid {
        entity.Description = dbEntity.Description.String
    }

    // Nullable timestamp
    if dbEntity.UpdatedAt.Valid {
        entity.UpdatedAt = &dbEntity.UpdatedAt.Time
    }

    // JSONB
    if len(dbEntity.Metadata) > 0 {
        var metadata map[string]interface{}
        if err := json.Unmarshal(dbEntity.Metadata, &metadata); err == nil {
            entity.Metadata = metadata
        }
    }

    return entity
}
```

### DB → Model (Slice)

```go
func (s *Service) toModels(dbEntities []db.Entity) []*model.Entity {
    models := make([]*model.Entity, len(dbEntities))
    for i, entity := range dbEntities {
        models[i] = s.toModel(entity)
    }
    return models
}
```

### Model → DB Params

```go
func (s *Service) Create(ctx, input model.CreateInput) (*model.Entity, error) {
    // Convert model → db params
    metadataJSON, _ := json.Marshal(input.Metadata)

    var description *string
    if input.Description != "" {
        description = &input.Description
    }

    dbEntity, err := s.queries.Create(ctx, db.CreateParams{
        ID:          uuid.New(),
        Name:        input.Name,
        Description: description,
        Metadata:    metadataJSON,
    })

    return s.toModel(dbEntity), nil
}
```

## 📋 Handler Template

```go
package entity

import (
    "doit/internal/model"
    "doit/internal/service"
    "doit/internal/web"
    "doit/pkg/logger"
    "net/http"
    "strconv"
    "github.com/google/uuid"
)

type Handler struct {
    log     *logger.Logger
    service *service.EntityService
}

func NewHandler(log *logger.Logger, service *service.EntityService) *Handler {
    return &Handler{log: log, service: service}
}

func (h *Handler) Create(w http.ResponseWriter, r *http.Request) error {
    var input model.CreateInput
    if err := web.DecodeJSON(r, &input); err != nil {
        return err
    }

    entity, err := h.service.Create(r.Context(), input)
    if err != nil {
        return err
    }

    return web.RespondCreated(w, r, entity)
}

func (h *Handler) Get(w http.ResponseWriter, r *http.Request) error {
    id, err := uuid.Parse(web.Param(r, "id"))
    if err != nil {
        return err
    }

    entity, err := h.service.GetByID(r.Context(), id)
    if err != nil {
        return err
    }

    return web.RespondOK(w, r, entity)
}

func (h *Handler) List(w http.ResponseWriter, r *http.Request) error {
    limit, _ := strconv.Atoi(r.URL.Query().Get("limit"))
    offset, _ := strconv.Atoi(r.URL.Query().Get("offset"))

    if limit == 0 {
        limit = 20
    }

    entities, err := h.service.List(r.Context(), int32(limit), int32(offset))
    if err != nil {
        return err
    }

    return web.RespondOK(w, r, entities)
}

func (h *Handler) Update(w http.ResponseWriter, r *http.Request) error {
    id, err := uuid.Parse(web.Param(r, "id"))
    if err != nil {
        return err
    }

    var input model.UpdateInput
    if err := web.DecodeJSON(r, &input); err != nil {
        return err
    }

    entity, err := h.service.Update(r.Context(), id, input)
    if err != nil {
        return err
    }

    return web.RespondOK(w, r, entity)
}

func (h *Handler) Delete(w http.ResponseWriter, r *http.Request) error {
    id, err := uuid.Parse(web.Param(r, "id"))
    if err != nil {
        return err
    }

    err = h.service.Delete(r.Context(), id)
    if err != nil {
        return err
    }

    return web.RespondNoContent(w, r)
}
```

## 🎯 Key Principles

1. **Models** = Clean domain types
2. **Services** = All business logic
3. **Handlers** = HTTP only
4. **Converters** = In services (private)
5. **Transactions** = In services
6. **Validation** = In services
